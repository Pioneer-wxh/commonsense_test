    if 'answer' in adata['output_pred']:
        ff = 1
        adata['output_pred'] = adata['output_pred'].split('answer')[1]
    if 'boxed' in adata['output_pred']:
        ff = 1
        adata['output_pred'] = adata['output_pred'].split('boxed')[1]
    if ff:
        sentence = adata['output_pred'].replace(',', '')
        pred = [s for s in re.findall(r'-?\d+\.?\d*', sentence)]
        pred_answer = float('inf')
        if len(pred) > 0:
            pred_answer = float(pred[0])
    else:
        sentence = adata['output_pred'].replace(',', '')
